## Bridge gotcha (so we don't repeat it)

The Webflow Designer MCP bridge times out when its tab isn't the foreground/active window. **A timeout means "unknown outcome" — re-list before retrying any write.** Mitigations: keep the Designer in a visible side-by-side window (not minimized, tab in front); Chrome → Settings → Performance → Memory Saver **off** (or add `webflow.com` to "always keep active"); `chrome://flags` → disable "Throttle Javascript timers in background" + "Quick intensive timer throttling of loaded background pages", relaunch. The embed kit avoids the bridge entirely.

# Webflow build kit (the plan to execute tomorrow)

**Division of labor:** you build the page + card in the Designer (full visual + responsive control); the embed script plugs in the live data.

## Step 1 — Build the page shell, set these IDs
(Select element → Settings panel / gear → ID)

| Element | ID |
|---|---|
| Text input (search field) | `ask-input` |
| Ask button | `ask-submit` |
| Answer text block | `ask-answer` |
| Empty container for cards (the bento) | `ask-sources` |

## Step 2 — Build ONE card as the template
Inside `#ask-sources`, add a **Link Block** → class `ask-card`, **ID `ask-card-template`** (leave its link empty; the script sets it). Inside it add these, each marked with a **Custom attribute** (Settings → Custom attributes → Name `data-ask`, Value as shown):

```
Link Block   .ask-card   #ask-card-template
├─ Image            data-ask = image      (upload any placeholder)
├─ Div  .ask-card-body   (position where you want, e.g. bottom)
│   ├─ Text "Source title"   data-ask = title
│   └─ Text "Type"           data-ask = kind
└─ Text "[1]"        data-ask = n         (badge, e.g. top-right)
```
Style it however you like (radius, padding, fonts, badge, body position) — you own the visual. The embed CSS only adds the bento grid, the per-type image fit (cover / contained-logo / monogram), and a gradient on photo cards; all overridable. The template card shows on the Designer canvas (handy for styling) and is auto-hidden on the live site — don't manually hide it.

## Step 3 — Add an HTML Embed element, paste this whole block

```html
<!-- Ask Z47 — wiring + bento CSS -->
<style>
  #ask-card-template{display:none !important;}
  #ask-sources{
    display:grid;
    grid-template-columns:repeat(auto-fill,minmax(150px,1fr));
    grid-auto-rows:124px; grid-auto-flow:dense; gap:10px;
  }
  @media (max-width:767px){
    #ask-sources{grid-template-columns:repeat(auto-fill,minmax(130px,1fr)); grid-auto-rows:108px; gap:8px;}
  }
  .ask-card{position:relative; overflow:hidden;}
  .ask-card.is-wide{grid-column:span 2;}
  .ask-card.is-feature{grid-column:span 2; grid-row:span 2;}
  .ask-card [data-ask="image"]{position:absolute; inset:0; width:100%; height:100%; object-fit:cover;}
  .ask-card [data-ask="title"], .ask-card [data-ask="kind"], .ask-card [data-ask="n"]{position:relative; z-index:2;}
  .ask-card.is-img::after{content:""; position:absolute; inset:0; z-index:1;
    background:linear-gradient(180deg,rgba(0,0,0,0) 40%,rgba(0,0,0,.7));}
  .ask-card.is-img [data-ask="title"], .ask-card.is-img [data-ask="kind"]{color:#fff;}
  .ask-card.is-logo{background:#fff;}
  .ask-card.is-logo [data-ask="image"]{object-fit:contain; padding:12px;}
  .ask-card.is-mono{background:#fff7f1;}
  .ask-card.is-mono [data-ask="image"]{display:none;}
  .ask-card.is-mono::before{content:"Z47"; position:absolute; top:11px; left:13px; z-index:2; font-weight:800; color:#e8702a;}
</style>
<script>
(function(){
  var EP = "/app/api/ask"; // same-origin: the app is mounted at /app on this site
  var KIND = {team:"Team",podcast:"Podcast",news:"News",portfolio:"Portfolio",faq:"FAQ",about:"About Z47"};
  var input=document.getElementById("ask-input"), btn=document.getElementById("ask-submit"),
      ans=document.getElementById("ask-answer"), grid=document.getElementById("ask-sources"),
      tpl=document.getElementById("ask-card-template");
  if(!input || !grid || !tpl) return;
  function render(s){
    var card=tpl.cloneNode(true); card.removeAttribute("id");
    var kind=s.kind||"about";
    var mode=s.image ? (kind==="portfolio" ? "is-logo" : "is-img") : "is-mono";
    card.classList.add("ask-card", mode);
    if(kind==="podcast"||kind==="news") card.classList.add("is-wide");
    if(mode==="is-img" && s.n===1) card.classList.add("is-feature");
    card.setAttribute("href", s.url); card.setAttribute("target","_blank"); card.setAttribute("rel","noopener");
    var img=card.querySelector('[data-ask="image"]');
    if(img){ img.removeAttribute("srcset"); img.removeAttribute("sizes"); if(s.image){ img.src=s.image; img.alt=s.title||""; } }
    var t=card.querySelector('[data-ask="title"]'); if(t) t.textContent=s.title||"";
    var k=card.querySelector('[data-ask="kind"]');  if(k) k.textContent=KIND[kind]||"";
    var n=card.querySelector('[data-ask="n"]');     if(n) n.textContent="["+s.n+"]";
    grid.appendChild(card);
  }
  function ask(){
    var q=(input.value||"").trim(); if(!q) return;
    if(ans) ans.textContent="Thinking…"; grid.innerHTML="";
    fetch(EP,{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({question:q})})
      .then(function(r){return r.json();})
      .then(function(d){
        if(ans) ans.textContent=d.answer||d.error||"No answer.";
        (d.sources||[]).filter(function(s){return s.url;}).forEach(render);
      })
      .catch(function(){ if(ans) ans.textContent="Network error. Please try again."; });
  }
  if(btn) btn.addEventListener("click", function(e){ e.preventDefault(); ask(); });
  input.addEventListener("keydown", function(e){ if(e.key==="Enter"){ e.preventDefault(); ask(); } });
})();
</script>
```
